3. Step-by-Step Implementation: The Python Webhook Listener

To securely dispatch articles from your blog into n8n without exposing sensitive API keys in the browser, deploy a lightweight Python webhook worker. Below is the battle-tested script using fastapi and requests that validates payloads and relays them to your local or VPS n8n instance:

# webhook_dispatcher.py — Production Webhook Relay for AIAndBeyond
import hmac
import hashlib
import os
import requests
from fastapi import FastAPI, HTTPException, Header, Request

app = FastAPI(title="Social Content Pipeline Webhook")

WEBHOOK_SECRET = os.getenv("PUBLISH_WEBHOOK_SECRET", "super-secure-shared-key-2026")
N8N_WEBHOOK_URL = os.getenv("N8N_INCOMING_URL", "http://localhost:5678/webhook/blog-published")

def verify_signature(payload_bytes: bytes, signature_header: str) -> bool:
    if not signature_header:
        return False
    computed = hmac.new(WEBHOOK_SECRET.encode(), payload_bytes, hashlib.sha256).hexdigest()
    return hmac.compare_digest(f"sha256={computed}", signature_header)

@app.post("/webhook/publish")
async def handle_publish_event(request: Request, x_hub_signature: str = Header(None)):
    body = await request.body()
    
    # 1. Verify cryptographic HMAC signature
    if not verify_signature(body, x_hub_signature):
        raise HTTPException(status_code=401, detail="Invalid webhook signature")
    
    payload = await request.json()
    article_title = payload.get("title")
    article_url = payload.get("url")
    article_markdown = payload.get("content", "")
    
    # 2. Forward sanitized payload to n8n orchestration engine
    n8n_payload = {
        "title": article_title,
        "url": article_url,
        "category": payload.get("category", "Tech"),
        "raw_content": article_markdown[:8000], # Enforce prompt safety buffer
        "source": "AIAndBeyond CMS"
    }
    
    response = requests.post(N8N_WEBHOOK_URL, json=n8n_payload, timeout=10)
    if response.status_code != 200:
        raise HTTPException(status_code=502, detail="n8n engine failed to accept webhook")
        
    return {"status": "success", "message": f"Article '{article_title}' queued for social repurposing"}
